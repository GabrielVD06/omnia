/** * Actualiza los datos del dashboard con una animación suave de opacidad * @param {string} systemId - ID de la tarjeta seleccionada */ const updateDashboardData = (systemId) => { const data = dashboardData[systemId]; if (!data) return; const dashBody = document.querySelector('.dash-body'); if (!dashBody) return; // Animación de salida (Fade out) para evitar saltos bruscos dashBody.style.opacity = '0.4'; dashBody.style.transition = 'opacity 0.15s ease'; setTimeout(() => { // Reemplazo dinámico de métricas clave const values = dashBody.querySelectorAll('.metric-value'); const trends = dashBody.querySelectorAll('.metric-trend, .metric-sub'); if (values.length >= 4) { values[0].textContent = data.ventas; values[1].textContent = data.pedidos; values[2].textContent = data.ocupacion; values[3].textContent = data.tiempo; } if (trends.length >= 4) { trends[0].textContent = data.ventasTrend; trends[1].textContent = data.pedidosSub; trends[2].textContent = data.ocupacionSub; trends[3].textContent = data.tiempoTrend; } // Animación de entrada (Fade in) dashBody.style.opacity = '1'; }, 150); }; // Asignación de eventos de interacción systemButtons.forEach(button => { button.addEventListener('click', (e) => { const currentButton = e.currentTarget;
